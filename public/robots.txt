# Staging. Keeping it out of search is done by the X-Robots-Tag header
# (noindex, nofollow, noarchive — see vercel.json) and the per-page robots
# meta tag, both of which are authoritative. A blanket Disallow here adds
# nothing to that, and it actively breaks link previews: Facebook, WhatsApp
# and LinkedIn all honour robots.txt in their link scrapers, so a shared
# URL rendered no card at all.
#
# So: everything stays disallowed for general crawlers, and the link-preview
# bots are named explicitly.

User-agent: *
Disallow: /
